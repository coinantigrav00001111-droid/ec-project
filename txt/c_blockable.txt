class_name C_Blockable extends IBlockable

##################################
## Библиотека действий. Она живет на объекте всегда.
##################################
## способность блочить ентити, когда включено C_Blocked

var areas: Array[Area3D] = []
var overlaps := 0



func on_added():
	var visual = entity.get_comp(C_Visual)
	if visual and not visual.static_nodes.is_empty():
		# Подключаемся ко ВСЕМ Area3D (и балласт, и рельсы, если они Area3D)
		# Это исключает баг, когда мы цепляемся за мелкий сокет
		for node in visual.static_nodes.keys():
			if node is Area3D:
				node.body_entered.connect(_on_enter)
				node.body_exited.connect(_on_exit)
				areas.append(node)
		
		_check_initial_overlaps()



func _check_initial_overlaps():
	for a in areas:
		for body in a.get_overlapping_bodies():
			_on_enter(body)



## ⚡Сигнал что есть вход боди
func _on_enter(body):
	if _should_block(body): # Надо ли делать блок?
		overlaps += 1
		_update_state()



## Обновление стейта при входе и выходе
func _update_state():
	var is_blocked = entity.has_comp(C_Blocked)
	var should_be_blocked = overlaps > 0

	if should_be_blocked and not is_blocked:
		entity.add_component(C_Blocked.new()) # Вешаем метку
		printt('block')
	elif not should_be_blocked and is_blocked:
		printt('unblock')
		entity.remove_component(C_Blocked)    # Снимаем метку



## Надо ли делать блок
func _should_block(body):
	var ent = find_entity_from_collider(body) # Выдает сущность
	if not ent or ent == entity: return false # Игнорим себя
	if ent.has_comp(C_Ghost): return false    # Игнорим гостов
	return true



## Из коллайдера поднимает его Entity
func find_entity_from_collider(node: Node) -> Entity:
	var current = node
	while current:
		if current is Entity:
			return current
		current = current.get_parent()
	
	return null



## ⚡Сигнал что есть выход из боди
func _on_exit(body):
	if _should_block(body):
		overlaps -= 1
		_update_state()



## При удалении компа чистим коннекты
func on_removed():
	for a in areas:
		if a:
			a.body_entered.disconnect(_on_enter)
			a.body_exited.disconnect(_on_exit)
	areas.clear()
		
	if entity.has_comp(C_Blocked):
		entity.remove_component(C_Blocked)
