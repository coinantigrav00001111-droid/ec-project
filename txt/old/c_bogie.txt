#### Управляет тележками, сканирует Слой 3 (Траки) рейкастами,

class BogieData:
	var id: int
	var config: BogieConfig # конфиг конкретного боги из identity
	var pivot: Node3D # корневая нода в сцене
	var raycast: RayCast3D # Датчик
	var current_track: Entity
	var track_distance: float
	var global_pos: Vector3
	var yaw_rotation: float
	var is_front: bool = false # Флаг для быстрого поиска
	

# Сами боги, их id, конфиги из TrainVisual перекидываются сюда
var bogies_data: Array[BogieData] = []

var _is_derailed: bool = false



func on_added() -> void:
	#Собираем массив данных боги
	var identity = entity.get_comp(C_Identity)
	var bogie_configs
	if identity:
		bogie_configs = identity.get_config('bogie_configs')	

	# Принимаем контроль боги из визуала
	var comp = entity.get_comp(C_TrainVisual)
	var bogie_nodes = comp.bogie_nodes
	
	# Формируем массив данных боги 
	for i in range(bogie_nodes.size()):
		var bogie = bogie_nodes[i]
		_create_bogies_data(i, bogie_configs[i], bogie)



func _create_bogies_data(_id: int, _config: BogieConfig, bogie:Node3D):
	
	var data = BogieData.new()
	data.id = _id
	data.config = _config
	data.pivot = bogie
	data.raycast = bogie.get_node('RayCast3D')
	# TODO а если больше 2х боги?
	var _offset_z = bogie.position.z
	data.is_front = _offset_z > 0.0
	bogies_data.append(data)



## Пересчет позиции в chassis 
func calculate_next_positions(delta: float, speed: float) -> void:
	for bogie in bogies_data:
		_update_bogie_math(bogie, delta, speed)



func _update_bogie_math(bogie: BogieData, delta: float, speed: float) -> void:
	if not bogie.raycast.is_colliding():
		# Тележка повисла в воздухе (сохраняем последнюю позицию или триггерим сход)
		return
		
	var collider = bogie.raycast.get_collider()
	var track_entity = _find_entity_from_collider(collider) 
	
	
	if not track_entity or not track_entity.has_comp(C_Path):
		return
		
	var path_comp = track_entity.get_comp(C_Path)
	var power_track = track_entity.get_comp(C_PowerTrack)
	
	# Учитываем инверсию кривой (is_flipped)
	var direction_multiplier = 1.0
	if power_track and power_track.is_flipped:
		direction_multiplier = -1.0 #
		
	# Шагаем по математической кривой
	bogie.track_distance += speed * delta * direction_multiplier
	
	# Вычисляем идеальную точку в мировых координатах
	var spline_pos_local = path_comp.curve.sample_baked(bogie.track_distance)
	bogie.global_pos = path_comp.path_node.to_global(spline_pos_local)
	
	# (Здесь же можно посчитать YAW-поворот для самой ноды bogie.pivot)



# API для C_Chassis
func get_front_position() -> Vector3:
	for b in bogies_data:
		if b.is_front: return b.global_pos
	return Vector3.ZERO

func get_rear_position() -> Vector3:
	for b in bogies_data:
		if not b.is_front: return b.global_pos
	return Vector3.ZERO

func _find_entity_from_collider(node: Node) -> Entity:
	var current = node
	while current:
		if current is Entity:
			return current
		current = current.get_parent()
	
	return null
