## Вызов из ентити при создании комп
func on_added():
	instance = scene.instantiate() # Создаем сцену
	entity.add_child(instance)

	_collect_meshes()
	_collect_collision_nodes()
	
	# instance.collide_with_areas = true нет такой для ригид
	# достраиваем боги
	bogie_configs = get_comp(C_Identity).get_config('bogie_configs')
	var offset_mesh_y = get_comp(C_Identity).get_config('offset_mesh_y')
	
	for bogie in bogie_configs: 
		
		var b_scene = bogie.scene.instantiate() # Создаем сцену
		
		b_scene.position.y = bogie.wheel_radius # боги  от пола вверх до центра колеса
		b_scene.position.z = bogie.offset.x 
		
		# Перестраиваем коллизию вагона с учетом габаритов с боги и колесами.
		var bogie_collision = b_scene.get_node('BogieCollisionShape3D')
		b_scene.remove_child(bogie_collision)
		bogie_collision.owner = null

		bogie_collision.position = b_scene.position # коллизию в точку боги
		
		instance.add_child(bogie_collision) #коллизия боги теперь у ригид кузова 
		# Назначаем новым владельцем корень текущей рабочей сцены
		bogie_collision.owner = instance.owner 
		bogie_collision.name = 'BogieCollisionShape3D_01'
		
		#НЕТ!  Перекидываем боги в корень ентити, больше визуал его не контролит
		#b_scene.owner = null
		#entity.add_child(b_scene) # боги и колеса
		instance.add_child(b_scene) # боги и колеса
		
		b_scene.name = 'Bogie_01'
		
		# поднимаем кузов 
		for mesh in mesh_nodes:
			#mesh.position.y = b_scene.position.y + bogie.offset.y
			mesh.position.y = offset_mesh_y # из вагон-конфига
			var wagon_collision = instance.get_node('WagonCollisionShape3D')
			wagon_collision.position.y = mesh.position.y + wagon_collision.shape.size.y / 2
			
	
			bogie_nodes.append(b_scene)

func on_select_layer(_data):

	if _data.enabled:
		instance.collision_layer =  Global.active_layers(['ghost'])
		instance.collision_mask = Global.active_layers(['ground','tracks','trains'])

	else: # возврат к дефолту, когда снимается селект
		instance.collision_layer = Global.active_layers(['trains'])
		# 1 << 4 	# заменяет активный слой на 5

## Смотрит удаляемый комп, если увидит, то сбросит в нормальный вид

func update_visual(_removing_type: Script = null): 
	if has_comp(C_Blocked) and _removing_type != C_Blocked:
		_set_blocked() 
	elif has_comp(C_Ghost) and _removing_type != C_Ghost:
		_set_ghost()
	elif has_comp(C_Selected) and _removing_type != C_Selected:
		_set_selected() 
	elif has_comp(C_Hovered) and _removing_type != C_Hovered:
		_set_hovered()
	else:
		_reset_overlays() # Сбрасываем ТОЛЬКО глобальные оверлеи
		


# Очищает глобальную пленку (Hover, Selected, Ghost и т.д.)
func _reset_overlays():
	for mesh in mesh_nodes:
		mesh.set_material_overlay(null)




func get_visual_offset() -> Vector3:
	# Вместо математики AABB, лучше просто возвращать Vector3.ZERO, 
	# если мы хотим держать рельс за его начало (Socket 0).
	return Vector3.ZERO



## Центр ентити по первому мешу
func get_center_world() -> Vector3:
	var mesh = mesh_nodes[0]
	var aabb = mesh.get_aabb()
	var local_center = aabb.size*0.5
	# По Х меш не смещен, поэтому его в 0
	local_center.x = 0 
	local_center.y = 0 
	return instance.to_global(local_center)	



## -----MESH-------

func _collect_meshes():
	mesh_nodes = [] 
	_find_meshes(instance) 


func _find_meshes(node): 
	if node is MeshInstance3D: 
		mesh_nodes.append(node)

	for child in node.get_children(): 
		_find_meshes(child)



## -----Сбор нод с коллизиями (Area3D, StaticBody3D, RigidBody3D)-------

func _collect_collision_nodes():
	collision_nodes = []
	_find_collisions(instance)


## Заполняем массив коллизий
func _find_collisions(node): 
	if node is CollisionObject3D: 
		collision_nodes.append(node)

	for child in node.get_children(): 
		_find_collisions(child)
