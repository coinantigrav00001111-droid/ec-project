## !!!!! Это фрагмент старого скрипта для сравнения !!!!

func generate_ballast_collisions(_parent_node, _debug_color=Color("00b929ff")) -> void: 
	var segment_props = GM.road_setting.track_ballast_segment
	
	var l_arc = (object_setting.RADIUS + segment_props.offset.x) * deg_to_rad(object_setting.DEG) #  длина  дуги
	# длина по Z
	var _segment_size_z = (l_arc - segment_props.distance * (segment_props.count - 1) - segment_props.offset.z * 2)/ segment_props.count
	
	var _d = 0.0 # текущий угол для создания коллизии
	var _segment_last = -1 # флаг последней коллизии
	var _segment_size = Vector3(segment_props.size.x, segment_props.size.y, _segment_size_z + segment_props.offset.z) #размер фрагмента с учетом всего 
	
	for i in range(0, segment_props.count):
		# угол для коллизии
		# по умолчанию _d = 0 и используется когда i=0, в остальном :
		#XX ====
		if i != 0: # первая коллизия делается от 0 вверх
			# сам угол + половина коллизии(для центра)
			_d = object_setting.DEG * (i+0.5) / float(segment_props.count)
			if i == segment_props.count - 1: # последняя коллизия  от полного угла вниз
				_d = object_setting.DEG
				_segment_last = 1 
				
		else: # i= 0
			#если приспичит править то тут и ниже есть
			#object_setting.DEG * (i+0.5) / float(segment_props.count)
			pass
		# Вместо лямбды сюда
		var new_segment = CollisionShape3D.new()
		_parent_node.add_child(new_segment)
		new_segment.shape = BoxShape3D.new()
		new_segment.shape.set_size(_segment_size)
		
		# трансформ для коллижн 
		var s_z = 0.0 # смещение от центра последней или первой отличается от остальных
		
		#Если приспичит править, то то что выше и тут 
		if _d == 0.0 or _segment_last == 1:# первая и последняя смещаются
			s_z = _segment_last * (_segment_size.z /2 - segment_props.offset.z)
		
		var _segment_tr = Transform3D(Basis.IDENTITY,Vector3(-object_setting.RADIUS - segment_props.offset.x,0,s_z))
		# поворот на угол
		_segment_tr = _segment_tr.rotated(Vector3.UP, deg_to_rad(-_d)) 

		## возврат на место
		_segment_tr.origin -= Vector3(-object_setting.RADIUS, -segment_props.size.y / 2, 0)
		new_segment.set_transform(_segment_tr)
		new_segment.set_debug_color(_debug_color)
